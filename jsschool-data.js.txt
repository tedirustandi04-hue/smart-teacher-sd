    initFormEvents() {
        const form = document.getElementById('school-profile-form');
        const alertEl = document.getElementById('profile-alert');

        if (form) {
            form.addEventListener('submit', (e) => {
                e.preventDefault();
                const formData = new FormData(form);
                const updatedData = {};
                
                formData.forEach((value, key) => {
                    updatedData[key] = value;
                });

                if (this.saveProfile(updatedData)) {
                    alertEl.className = "p-4 mb-4 rounded-lg text-sm font-medium bg-emerald-50 text-emerald-800 border border-emerald-200 flex items-center";
                    alertEl.innerHTML = `<i data-lucide="check-circle" class="w-5 h-5 mr-2 text-emerald-600"></i> Profil Sekolah berhasil diperbarui dan disimpan!`;
                    alertEl.classList.remove('hidden');
                    lucide.createIcons();

                    setTimeout(() => {
                        alertEl.classList.add('hidden');
                    }, 4000);
                }
            });
        }
    }
