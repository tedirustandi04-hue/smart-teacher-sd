    static initEvents() {
        const btnFetch = document.getElementById('btn-fetch-cp');
        if (btnFetch) {
            btnFetch.addEventListener('click', () => {
                const subject = document.getElementById('cp-subject').value;
                const phase = document.getElementById('cp-phase').value;

                document.getElementById('preview-subject').textContent = subject;
                document.getElementById('preview-phase').textContent = phase;

                const db = window.CP_REFERENCE_DATABASE[subject]?.[phase] || [
                    { element: "Umum", content: "Capaian Pembelajaran standar belum dikonfigurasi untuk kombinasi mata pelajaran dan fase ini." }
                ];

                let html = "";
                db.forEach((item, index) => {
                    html += `
                        <div class="border border-slate-300 p-3 rounded">
                            <p class="font-bold text-blue-900 border-b pb-1 mb-2">${index + 1}. Elemen: ${item.element}</p>
                            <p class="text-justify">${item.content}</p>
                        </div>
                    `;
                });

                document.getElementById('cp-content-area').innerHTML = html;
                // Simpan ke storage lokal agar bisa ditarik oleh TP Generator
                localStorage.setItem('active_cp_data', JSON.stringify({ subject, phase, elements: db }));
            });
        }
    }
}

window.ToolCP = ToolCP;
