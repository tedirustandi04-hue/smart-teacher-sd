        if (btnSync) {
            btnSync.addEventListener('click', () => {
                const passRateEl = document.getElementById('stat-pass-rate');
                if (passRateEl) {
                    const rateText = passRateEl.textContent.replace('%', '');
                    document.getElementById('followup-passrate').value = parseInt(rateText) || 60;
                    alert("Data persentase ketuntasan berhasil ditarik dari Mesin Analisis Soal!");
                } else {
                    alert("Data analisis soal belum dihitung. Jalankan Tool 17 (Analisis Soal) terlebih dahulu!");
                }
            });
