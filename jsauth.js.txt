// js/auth.js

class AuthService {
    constructor() {
        this.currentUser = this.getUserSession();
    }

    getUserSession() {
        const session = localStorage.getItem(CONFIG.STORAGE_KEYS.USER_SESSION);
        return session ? JSON.parse(session) : null;
    }

    isAuthenticated() {
        return this.currentUser !== null;
    }

    login(email, password) {
        return new Promise((resolve, reject) => {
            setTimeout(() => {
                if (email && password.length >= 6) {
                    const userData = {
                        id: "USR-" + Date.now(),
                        name: email.split('@')[0].replace('.', ' ').toUpperCase(),
                        email: email,
                        role: "Guru SD",
                        avatar: `https://ui-avatars.com/api/?name=${encodeURIComponent(email)}&background=0ea5e9&color=fff`
                    };
                    localStorage.setItem(CONFIG.STORAGE_KEYS.USER_SESSION, JSON.stringify(userData));
                    this.currentUser = userData;
                    this.updateUIProfile();
                    resolve({ success: true, user: userData });
                } else {
                    reject({ success: false, message: "Email atau kata sandi tidak valid (Min. 6 Karakter)." });
                }
            }, 600);
        });
    }

    register(name, email, password) {
        return new Promise((resolve, reject) => {
            setTimeout(() => {
                if (name && email && password.length >= 6) {
                    const userData = {
                        id: "USR-" + Date.now(),
                        name: name,
                        email: email,
                        role: "Guru SD",
                        avatar: `https://ui-avatars.com/api/?name=${encodeURIComponent(name)}&background=0ea5e9&color=fff`
                    };
                    localStorage.setItem(CONFIG.STORAGE_KEYS.USER_SESSION, JSON.stringify(userData));
                    this.currentUser = userData;
                    this.updateUIProfile();
                    resolve({ success: true, user: userData });
                } else {
                    reject({ success: false, message: "Lengkapi semua data dengan benar." });
                }
            }, 600);
        });
    }

    logout() {
        localStorage.removeItem(CONFIG.STORAGE_KEYS.USER_SESSION);
        this.currentUser = null;
        window.location.hash = "#login";
        this.updateUIProfile();
    }

    updateUIProfile() {
        const profileNameEl = document.getElementById('user-profile-name');
        const profileRoleEl = document.getElementById('user-profile-role');
        const profileAvatarEl = document.getElementById('user-profile-avatar');

        if (this.currentUser) {
            if (profileNameEl) profileNameEl.textContent = this.currentUser.name;
            if (profileRoleEl) profileRoleEl.textContent = this.currentUser.role;
            if (profileAvatarEl) profileAvatarEl.src = this.currentUser.avatar;
        } else {
            if (profileNameEl) profileNameEl.textContent = "Tamu (Belum Login)";
            if (profileRoleEl) profileRoleEl.textContent = "Klik untuk Login";
        }
    }
}

window.authService = new AuthService();
