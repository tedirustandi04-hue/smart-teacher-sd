// js/tool-bahan-ajar.js

class ToolBahanAjar {
    static render() {
        const school = window.schoolDataService.getProfile();
        return `
            <div class="space-y-6">
                <div class="bg-white p-6 rounded-xl border border-slate-200 shadow-sm flex justify-between items-center">
                    <div>
                        <h1 class="text-xl font-bold text-slate-800 flex items-center">
                            <i data-lucide="file-text" class="w-6 h-6 text-blue-600 mr-2"></i> Generator Bahan Ajar Siswa
                        </h1>
                        <p class="text-xs text-slate-500 mt-1">Menyusun bahan bacaan, ringkasan materi tematik, dan glosarium untuk peserta didik.</p>
                    </div>
                </div>

                <div class="grid grid-cols-1 lg:grid-cols-3 gap-6">
                    <div class="bg-white p-6 rounded-xl border border-slate-200 shadow-sm space-y-4 h-fit">
                        <h2 class="font-semibold text-slate-700 border-b pb-2 text-sm">Sumber Data Bahan Ajar</h2>
                        <button id="btn-sync-modul-bahan" class="w-full bg-slate-100 hover:bg-slate-200 text-slate-700 border border-slate-300 py-2 rounded-lg text-xs font-semibold flex items-center justify-center">
                            <i data-lucide="link" class="w-3.5 h-3.5 mr-2"></i> Sinkronkan dari Modul Ajar
                        </button>

                        <div>
                            <label class="block text-xs font-semibold text-slate-600 mb-1">Topik Utama</label>
                            <input type="text" id="bahan-topic" value="Sistem Pencernaan Manusia" class="w-full px-3 py-2 border border-slate-300 rounded-lg text-sm">
                        </div>

                        <button id="btn-generate-bahan" class="w-full bg-emerald-600 hover:bg-emerald-700 text-white py-2.5 rounded-lg text-sm font-semibold flex justify-center items-center shadow-sm">
                            <i data-lucide="sparkles" class="w-4 h-4 mr-2"></i> Buat Bahan Ajar
                        </button>
                    </div>

                    <div class="lg:col-span-2">
                        <div id="bahan-a4-preview" class="a4-preview text-slate-800">
                            <div class="text-center border-b-2 border-slate-800 pb-3 mb-6">
                                <h2 class="text-base font-bold uppercase">${school.schoolName}</h2>
                                <p class="text-xs">BAHAN BACAAN PESERTA DIDIK - SEMESTER ${school.semester}</p>
                            </div>

                            <div id="bahan-preview-body" class="space-y-4 text-xs">
                                <p class="italic text-slate-400">Klik "Buat Bahan Ajar" untuk menyusun materi ringkasan siswa...</p>
                            </div>
                        </div>
                    </div>
                </div>
            </div>
        `;
    }

    static initEvents() {
        const btnSync = document.getElementById('btn-sync-modul-bahan');
        const btnGen = document.getElementById('btn-generate-bahan');

        if (btnSync) {
            btnSync.addEventListener('click', () => {
                const activeModul = JSON.parse(localStorage.getItem('active_modul_data') || '{}');
                if (activeModul.identitas) {
                    document.getElementById('bahan-topic').value = activeModul.identitas.materiPokok;
                    alert("Topik berhasil disinkronkan dari Modul Ajar!");
                } else {
                    alert("Modul Ajar belum dibuat. Silakan buat Modul Ajar terlebih dahulu!");
                }
            });
        }

        if (btnGen) {
            btnGen.addEventListener('click', () => {
                const topic = document.getElementById('bahan-topic').value;
                const bahan = window.AIEngine.generateBahanAjar(topic);

                let html = `
                    <h3 class="font-bold text-sm text-center text-blue-900 border-b pb-2 mb-3">${bahan.judul}</h3>
                    <p class="leading-relaxed bg-blue-50/50 p-3 rounded border border-blue-100">${bahan.ringkasanMateri}</p>

                    <div class="space-y-3 pt-2">
                `;

                bahan.subBab.forEach(sub => {
                    html += `
                        <div>
                            <h4 class="font-bold text-slate-800">${sub.subJudul}</h4>
                            <p class="text-justify leading-relaxed mt-1">${sub.uraian}</p>
                        </div>
                    `;
                });

                html += `
                    </div>
                    <div class="pt-4 border-t mt-4">
                        <h4 class="font-bold mb-2">Glosarium / Kamus Kata Kunci:</h4>
                        <ul class="list-disc pl-5 space-y-1">
                            ${bahan.glosarium.map(g => `<li><strong>${g.istilah}:</strong> ${g.arti}</li>`).join('')}
                        </ul>
                    </div>
                `;

                document.getElementById('bahan-preview-body').innerHTML = html;
            });
        }
    }
}

window.ToolBahanAjar = ToolBahanAjar;
