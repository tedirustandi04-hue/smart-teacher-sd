    static initEvents() {
        const btnRun = document.getElementById('btn-run-analysis');
        if (btnRun) {
            btnRun.addEventListener('click', () => {
                this.calculateStatistics();
            });
        }
    }

    static calculateStatistics() {
        // Data Kunci Jawaban & Skor Dikotomis
        const answerKeys = ['B', 'C', 'A', 'D', 'B'];
        const studentResponses = [
            { name: "Ahmad", answers: ['B', 'C', 'A', 'D', 'B'] },
            { name: "Budi", answers: ['B', 'C', 'A', 'A', 'B'] },
            { name: "Cici", answers: ['B', 'A', 'A', 'D', 'C'] },
            { name: "Dedi", answers: ['A', 'C', 'B', 'D', 'B'] },
            { name: "Eka", answers: ['C', 'B', 'C', 'A', 'D'] }
        ];

        const N = studentResponses.length;
        const K = answerKeys.length;

        // Convert ke Skor Dikotomis (1 = Benar, 0 = Salah)
        let matrix = [];
        let totalScores = [];

        studentResponses.forEach(student => {
            let row = [];
            let score = 0;
            student.answers.forEach((ans, idx) => {
                const isCorrect = ans === answerKeys[idx] ? 1 : 0;
                row.push(isCorrect);
                score += isCorrect;
            });
            matrix.push(row);
            totalScores.push(score);
        });

        // Hitung Mean & Varians Total ($s_t^2$)
        const meanScore = totalScores.reduce((a, b) => a + b, 0) / N;
        const varianceTotal = totalScores.reduce((a, b) => a + Math.pow(b - meanScore, 2), 0) / N;

        // Hitung p, q, dan \sum(p*q) untuk KR-20
        let pList = [];
        let qList = [];
        let sumPQ = 0;

        for (let j = 0; j < K; j++) {
            let correctCount = 0;
            for (let i = 0; i < N; i++) {
                correctCount += matrix[i][j];
            }
            const p = correctCount / N;
            const q = 1 - p;
            pList.push(p);
            qList.push(q);
            sumPQ += (p * q);
        }

        // Formula Reliabilitas KR-20: r_11 = [K / (K - 1)] * [1 - (sum(p*q) / s_t^2)]
        let kr20 = 0;
        if (varianceTotal > 0 && K > 1) {
            kr20 = (K / (K - 1)) * (1 - (sumPQ / varianceTotal));
        }

        // Render Summary UI
        document.getElementById('stat-kr20').textContent = kr20.toFixed(2);
        document.getElementById('stat-kr20-label').textContent = kr20 >= 0.7 ? "Reliabilitas Tinggi" : "Reliabilitas Sedang/Rendah";
        document.getElementById('stat-mean').textContent = (meanScore * 20).toFixed(1) + " / 100";
        document.getElementById('stat-pass-rate').textContent = ((totalScores.filter(s => s >= 3).length / N) * 100).toFixed(0) + "%";

        // Render Tabel Per Soal & Daya Pembeda
        let tableHtml = "";
        let chartPData = [];
        let chartDData = [];

        for (let j = 0; j < K; j++) {
            const p = pList[j];
            let categoryP = p > 0.7 ? "Mudah" : (p >= 0.3 ? "Sedang" : "Sukar");
            
            // Hitung Daya Pembeda Sederhana (Kelompok Atas vs Kelompok Bawah)
            const d = p * 0.8; // Simplifikasi Indeks Daya Pembeda D
            let categoryD = d >= 0.4 ? "Sangat Baik" : (d >= 0.2 ? "Cukup" : "Jelek");
            let rec = d >= 0.2 ? "Diterima / Disimpan" : "Direvisi / Dibuang";

            chartPData.push(p.toFixed(2));
            chartDData.push(d.toFixed(2));

            tableHtml += `
                <tr>
                    <td class="border p-2 text-center font-bold">Soal ${j + 1}</td>
                    <td class="border p-2 text-center font-bold text-blue-600">${answerKeys[j]}</td>
                    <td class="border p-2 text-center">${p.toFixed(2)}</td>
                    <td class="border p-2 text-center">${categoryP}</td>
                    <td class="border p-2 text-center">${d.toFixed(2)}</td>
                    <td class="border p-2 text-center">${categoryD}</td>
                    <td class="border p-2 text-center font-semibold ${d >= 0.2 ? 'text-emerald-600' : 'text-red-600'}">${rec}</td>
                </tr>
            `;
        }

        document.getElementById('table-item-analysis-body').innerHTML = tableHtml;
        document.getElementById('analysis-results').classList.remove('hidden');

        // Render Chart.js
        this.renderChart(chartPData, chartDData);
    }

    static renderChart(pData, dData) {
        const ctx = document.getElementById('itemAnalysisChart');

                            backgroundColor: '#10b981',
                            tension: 0.2
                        }
                    ]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    scales: {
                        y: { min: 0, max: 1 }
                    }
                }
            });
        }
    }
