                const phase = document.getElementById('modul-phase').value;

                const topic = document.getElementById('modul-topic').value;
                const modelPembelajaran = document.getElementById('modul-model').value;

                // Ambil data TP jika tersimpan di localStorage
                const tps = JSON.parse(localStorage.getItem('active_tp_data') || '[]');
                const tpText = tps.length > 0 ? tps[0] : null;

                const modul = window.AIEngine.generateModulAjar({
                    subject, phase, classLevel, topic, tpText, modelPembelajaran
                });

                let html = `
                    <div class="bg-slate-50 p-3 rounded border border-slate-200 mb-4">

                            <tr><td class="font-semibold">Target Peserta Didik</td><td>: ${modul.targetPesertaDidik}</td></tr>
                        </table>
                    </div>

                            </ul>
