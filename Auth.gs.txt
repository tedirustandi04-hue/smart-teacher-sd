/**
 * Security & Auth Service Engine
 */

const Auth = {
  // Hashing password dengan SHA-256
  hashPassword(password) {
    const rawHash = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, password);
    return rawHash.map(b => (b < 0 ? b + 256 : b).toString(16).padStart(2, '0')).join('');
  },

  registerUser(email, password, name) {
    const existing = Database.findBy("USERS", "email", email);
    if (existing) {
      return { success: false, message: "Email sudah terdaftar!" };
    }

    const userId = "USR-" + Utilities.getUuid();
    const newRecord = {
      user_id: userId,
      email: email,
      password_hash: this.hashPassword(password),
      name: name,
      nip: "",
      role: "Guru SD",
      created_at: new Date().toISOString(),
      last_login: new Date().toISOString()
    };

    Database.insert("USERS", newRecord);
    return { success: true, user: { user_id: userId, email: email, name: name } };
  },

  loginUser(email, password) {
    const user = Database.findBy("USERS", "email", email);
    if (!user) return { success: false, message: "Email tidak ditemukan." };

    const hashedInput = this.hashPassword(password);
    if (user.password_hash !== hashedInput) {
      return { success: false, message: "Kata sandi salah!" };
    }

    // Update last login
    Database.update("USERS", "user_id", user.user_id, {
      last_login: new Date().toISOString()
    });

    const token = Utilities.base64Encode(user.user_id + ":" + new Date().getTime());

    return {
      success: true,
      token: token,
      user: {
        user_id: user.user_id,
        email: user.email,
        name: user.name,
        role: user.role
      }
    };
  }
};
